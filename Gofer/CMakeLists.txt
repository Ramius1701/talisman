cmake_minimum_required (VERSION 3.0)
project (gofer)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MODULE_PATH ${gofer_SOURCE_DIR}/../cmake/)
set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -Wall -ggdb")
set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -Wall")
find_package(SQLITE3 REQUIRED)
add_executable(gofer Gofer.cpp Request.cpp ../Common/Logger.cpp)
include_directories(${SQLITE3_INCLUDE_DIRS} ../Common)
if ("${CMAKE_CXX_COMPILER_ID}" STREQUAL "Clang")
	target_link_libraries (gofer ${SQLITE3_LIBRARIES})
else()
	if (CMAKE_SYSTEM_NAME MATCHES "Haiku")
		target_link_libraries (gofer ${SQLITE3_LIBRARIES} -lnetwork)
	elseif(CMAKE_SYSTEM_NAME MATCHES "Darwin")
		target_link_libraries (gofer  ${SQLITE3_LIBRARIES})
	else()
		target_link_libraries (gofer ${SQLITE3_LIBRARIES} -lstdc++fs)
	endif()
endif()
