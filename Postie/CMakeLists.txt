cmake_minimum_required (VERSION 3.0)
project (postie)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MODULE_PATH ${postie_SOURCE_DIR}/../cmake/)
set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -Wall -ggdb")
set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -Wall")
find_package(SQLITE3 REQUIRED)
add_executable(postie main.cpp Archiver.cpp Config.cpp ../Common/Logger.cpp Scanner.cpp ../Common/Squish.cpp Tosser.cpp Dupe.cpp TicProc.cpp)
include_directories(${SQLITE3_INCLUDE_DIRS} ../Common)
if ("${CMAKE_CXX_COMPILER_ID}" STREQUAL "Clang")
	target_link_libraries (postie ${SQLITE3_LIBRARIES})
else()
	if (CMAKE_SYSTEM_NAME MATCHES "Haiku")
		target_link_libraries (postie ${SQLITE3_LIBRARIES})
	elseif(CMAKE_SYSTEM_NAME MATCHES "Darwin")
		target_link_libraries (postie  ${SQLITE3_LIBRARIES})
	else()
		target_link_libraries (postie ${SQLITE3_LIBRARIES} -lstdc++fs)
	endif()
endif()
